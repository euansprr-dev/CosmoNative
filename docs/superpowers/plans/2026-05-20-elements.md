# Elements Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add reusable, collapsible, nestable Elements to shared CosmoOS rich-document editing surfaces, with slash-menu insertion and Sanctuary settings management.

**Architecture:** Store global Element definitions in an Application Support JSON store, and store inserted Element instances inside `RichDocument` blocks. The shared editor and renderer consume the same `RichDocument` model so note focus mode, content focus mode, and canvas note blocks inherit the behavior.

**Tech Stack:** Swift 5.9, SwiftUI, AppKit/TextKit, XCTest, Application Support JSON persistence, existing `DS` design tokens.

import Foundation

struct DocumentElementDefinition: Identifiable, Codable, Equatable, Hashable, Sendable {
    var id: UUID
    var title: String
    var systemIcon: String
    var createdAt: Date
    var updatedAt: Date
    var isDisabled: Bool

    init(
        id: UUID = UUID(),
        title: String,
        systemIcon: String = DocumentElementDefinition.defaultSystemIcon,
        createdAt: Date = Date(),
        updatedAt: Date = Date(),
        isDisabled: Bool = false
    ) {
        self.id = id
        self.title = Self.normalizedTitle(title)
        self.systemIcon = systemIcon.isEmpty ? Self.defaultSystemIcon : systemIcon
        self.createdAt = createdAt
        self.updatedAt = updatedAt
        self.isDisabled = isDisabled
    }

    static let defaultSystemIcon = "square.stack.3d.up"

    static func normalizedTitle(_ raw: String) -> String {
        raw
            .replacingOccurrences(of: #"\s+"#, with: " ", options: .regularExpression)
            .trimmingCharacters(in: .whitespacesAndNewlines)
    }

    var searchText: String {
        "\(title) \(systemIcon)"
            .lowercased()
            .replacingOccurrences(of: #"[^a-z0-9 ]+"#, with: " ", options: .regularExpression)
            .replacingOccurrences(of: #"\s+"#, with: " ", options: .regularExpression)
            .trimmingCharacters(in: .whitespacesAndNewlines)
    }
}

struct RichElementInstance: Identifiable, Codable, Equatable, Hashable, Sendable {
    var id: UUID
    var definitionID: UUID?
    var titleSnapshot: String
    var iconSnapshot: String
    var isCollapsed: Bool

    init(
        id: UUID = UUID(),
        definitionID: UUID?,
        titleSnapshot: String,
        iconSnapshot: String,
        isCollapsed: Bool = false
    ) {
        self.id = id
        self.definitionID = definitionID
        self.titleSnapshot = DocumentElementDefinition.normalizedTitle(titleSnapshot)
        self.iconSnapshot = iconSnapshot.isEmpty ? DocumentElementDefinition.defaultSystemIcon : iconSnapshot
        self.isCollapsed = isCollapsed
    }
}

@MainActor
final class DocumentElementStore: ObservableObject {
    static let shared = DocumentElementStore()

    @Published private(set) var definitions: [DocumentElementDefinition]

    private let fileURL: URL

    var activeDefinitions: [DocumentElementDefinition] {
        definitions
            .filter { !$0.isDisabled }
            .sorted { lhs, rhs in
                lhs.title.localizedCaseInsensitiveCompare(rhs.title) == .orderedAscending
            }
    }

    init(fileURL: URL = DocumentElementStore.defaultFileURL()) {
        self.fileURL = fileURL
        self.definitions = Self.loadDefinitions(from: fileURL)
    }

    @discardableResult
    func createDefinition(title: String, systemIcon: String) throws -> DocumentElementDefinition {
        let normalized = DocumentElementDefinition.normalizedTitle(title)
        let definition = DocumentElementDefinition(
            title: normalized.isEmpty ? "Untitled Element" : normalized,
            systemIcon: systemIcon
        )
        definitions.append(definition)
        try persist()
        return definition
    }

    @discardableResult
    func updateDefinition(id: UUID, title: String, systemIcon: String) throws -> DocumentElementDefinition {
        guard let index = definitions.firstIndex(where: { $0.id == id }) else {
            throw DocumentElementStoreError.definitionNotFound
        }
        definitions[index].title = DocumentElementDefinition.normalizedTitle(title).isEmpty
            ? "Untitled Element"
            : DocumentElementDefinition.normalizedTitle(title)
        definitions[index].systemIcon = systemIcon.isEmpty
            ? DocumentElementDefinition.defaultSystemIcon
            : systemIcon
        definitions[index].updatedAt = Date()
        try persist()
        return definitions[index]
    }

    func disableDefinition(id: UUID) throws {
        guard let index = definitions.firstIndex(where: { $0.id == id }) else {
            throw DocumentElementStoreError.definitionNotFound
        }
        definitions[index].isDisabled = true
        definitions[index].updatedAt = Date()
        try persist()
    }

    func search(_ query: String) -> [DocumentElementDefinition] {
        let normalized = Self.normalizedSearch(query)
        guard !normalized.isEmpty else { return activeDefinitions }
        return activeDefinitions
            .filter { $0.searchText.contains(normalized) }
            .sorted { lhs, rhs in
                Self.score(lhs, query: normalized) > Self.score(rhs, query: normalized)
            }
    }

    private func persist() throws {
        try FileManager.default.createDirectory(
            at: fileURL.deletingLastPathComponent(),
            withIntermediateDirectories: true
        )
        let encoder = JSONEncoder()
        encoder.outputFormatting = [.prettyPrinted, .sortedKeys]
        encoder.dateEncodingStrategy = .iso8601
        let data = try encoder.encode(definitions)
        try data.write(to: fileURL, options: .atomic)
    }

    private static func loadDefinitions(from fileURL: URL) -> [DocumentElementDefinition] {
        guard FileManager.default.fileExists(atPath: fileURL.path) else { return [] }
        do {
            let data = try Data(contentsOf: fileURL)
            let decoder = JSONDecoder()
            decoder.dateDecodingStrategy = .iso8601
            return try decoder.decode([DocumentElementDefinition].self, from: data)
        } catch {
            recoverCorruptFile(at: fileURL)
            return []
        }
    }

    private static func recoverCorruptFile(at fileURL: URL) {
        let formatter = ISO8601DateFormatter()
        let suffix = formatter.string(from: Date()).replacingOccurrences(of: ":", with: "-")
        let recoveredURL = fileURL.deletingLastPathComponent()
            .appendingPathComponent("Elements.corrupt-\(suffix).json")
        try? FileManager.default.moveItem(at: fileURL, to: recoveredURL)
        try? FileManager.default.createDirectory(
            at: fileURL.deletingLastPathComponent(),
            withIntermediateDirectories: true
        )
        try? Data("[]".utf8).write(to: fileURL, options: .atomic)
    }

    private static func defaultFileURL() -> URL {
        FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]
            .appendingPathComponent("CosmoOS", isDirectory: true)
            .appendingPathComponent("Elements.json")
    }

    private static func normalizedSearch(_ raw: String) -> String {
        raw.lowercased()
            .replacingOccurrences(of: #"[^a-z0-9 ]+"#, with: " ", options: .regularExpression)
            .replacingOccurrences(of: #"\s+"#, with: " ", options: .regularExpression)
            .trimmingCharacters(in: .whitespacesAndNewlines)
    }

    private static func score(_ definition: DocumentElementDefinition, query: String) -> Int {
        let title = normalizedSearch(definition.title)
        if title == query { return 100 }
        if title.hasPrefix(query) { return 80 }
        if definition.searchText.hasPrefix(query) { return 60 }
        if title.contains(query) { return 40 }
        if definition.searchText.contains(query) { return 20 }
        return 0
    }
}

enum DocumentElementStoreError: LocalizedError {
    case definitionNotFound

    var errorDescription: String? {
        switch self {
        case .definitionNotFound:
            return "Element definition not found."
        }
    }
}

struct RichBlock: Identifiable, Codable, Equatable, Hashable, Sendable {
    var id: UUID = UUID()
    var kind: RichBlockKind
    var inlines: [RichInlineNode] = []
    var checked: Bool? = nil
    var element: RichElementInstance? = nil
    var children: [RichBlock] = []

    enum CodingKeys: String, CodingKey {
        case id, kind, inlines, checked, element, children
    }

    init(
        id: UUID = UUID(),
        kind: RichBlockKind,
        inlines: [RichInlineNode] = [],
        checked: Bool? = nil,
        element: RichElementInstance? = nil,
        children: [RichBlock] = []
    ) {
        self.id = id
        self.kind = kind
        self.inlines = inlines
        self.checked = checked
        self.element = element
        self.children = children
    }

    init(from decoder: Decoder) throws {
        let container = try decoder.container(keyedBy: CodingKeys.self)
        self.id = try container.decodeIfPresent(UUID.self, forKey: .id) ?? UUID()
        self.kind = try container.decode(RichBlockKind.self, forKey: .kind)
        self.inlines = try container.decodeIfPresent([RichInlineNode].self, forKey: .inlines) ?? []
        self.checked = try container.decodeIfPresent(Bool.self, forKey: .checked)
        self.element = try container.decodeIfPresent(RichElementInstance.self, forKey: .element)
        self.children = try container.decodeIfPresent([RichBlock].self, forKey: .children) ?? []
    }

    func encode(to encoder: Encoder) throws {
        var container = encoder.container(keyedBy: CodingKeys.self)
        try container.encode(id, forKey: .id)
        try container.encode(kind, forKey: .kind)
        try container.encode(inlines, forKey: .inlines)
        try container.encodeIfPresent(checked, forKey: .checked)
        try container.encodeIfPresent(element, forKey: .element)
        if !children.isEmpty {
            try container.encode(children, forKey: .children)
        }
    }

    static func paragraph(_ text: String) -> RichBlock {
        RichBlock(kind: .paragraph, inlines: [.text(text)])
    }
}

private static func plainText(for block: RichBlock, at index: Int, siblings: [RichBlock]) -> String {
    switch block.kind {
    case .element:
        let title = block.element?.titleSnapshot.isEmpty == false ? block.element!.titleSnapshot : "Element"
        let childText = block.children.enumerated()
            .map { childIndex, child in
                plainText(for: child, at: childIndex, siblings: block.children)
            }
            .joined(separator: "\n")
        if childText.isEmpty {
            return "[\(title)]\n[/\(title)]"
        }
        return "[\(title)]\n\(childText)\n[/\(title)]"
    case .paragraph:
        return block.inlines.map(\.plainText).joined()
    case .heading1:
        return "# " + block.inlines.map(\.plainText).joined()
    case .heading2:
        return "## " + block.inlines.map(\.plainText).joined()
    case .heading3:
        return "### " + block.inlines.map(\.plainText).joined()
    case .quote:
        return "│ " + block.inlines.map(\.plainText).joined()
    case .divider:
        return "───────────────"
    case .bulletList:
        return "• " + block.inlines.map(\.plainText).joined()
    case .numberedList:
        var listPosition = 1
        var j = index - 1
        while j >= 0 && siblings[j].kind == .numberedList {
            listPosition += 1
            j -= 1
        }
        return "\(listPosition). " + block.inlines.map(\.plainText).joined()
    case .checklist:
        return ((block.checked ?? false) ? "☑ " : "☐ ") + block.inlines.map(\.plainText).joined()
    case .image:
        return "[Image]"
    }
}

        SlashCommand(type: .heading3, title: "Heading 3", subtitle: "Small section heading", icon: "textformat.size.smaller"),
        SlashCommand(type: .quote, title: "Quote", subtitle: "Add a block quote", icon: "text.quote"),
        SlashCommand(type: .divider, title: "Divider", subtitle: "Visual separation between sections", icon: "minus"),
        SlashCommand(type: .bulletList, title: "Bullet List", subtitle: "Create a bullet list", icon: "list.bullet"),
        SlashCommand(type: .numberedList, title: "Numbered List", subtitle: "Create a numbered list", icon: "list.number"),
        SlashCommand(type: .checkbox, title: "Checklist", subtitle: "Track tasks with checkboxes", icon: "checklist")
    ]

    static func filteredCommands(
        searchText: String,
        elements: [DocumentElementDefinition]
    ) -> [SlashCommand] {
        let elementCommands = elements.map { definition in
            SlashCommand(
                id: definition.id,
                type: .element,
                title: definition.title,
                subtitle: "Insert Element",
                icon: definition.systemIcon,
                elementDefinition: definition
            )
        }
        let allCommands = elementCommands + systemCommands
        let query = normalized(searchText)
        guard !query.isEmpty else { return allCommands }
        return allCommands
            .filter { command in
                Self.normalized(command.title).contains(query) ||
                Self.normalized(command.subtitle).contains(query)
            }
            .sorted { lhs, rhs in
                score(lhs, query: query) > score(rhs, query: query)
            }
    }

    private static func normalized(_ raw: String) -> String {
        raw.lowercased()
            .replacingOccurrences(of: #"[^a-z0-9 ]+"#, with: " ", options: .regularExpression)
            .replacingOccurrences(of: #"\s+"#, with: " ", options: .regularExpression)
            .trimmingCharacters(in: .whitespacesAndNewlines)
    }

    private static func score(_ command: SlashCommand, query: String) -> Int {
        let title = normalized(command.title)
        let subtitle = normalized(command.subtitle)
        if title == query { return 100 }
        if title.hasPrefix(query) { return 80 }
        if command.type == .element && title.contains(query) { return 70 }
        if subtitle.hasPrefix(query) { return 50 }
        if title.contains(query) { return 40 }
        if subtitle.contains(query) { return 20 }
        return 0
    }
}

private static func mergeCollapsedChildren(
    parsedBlocks: [RichBlock],
    previousBlocks: [RichBlock]
) -> [RichBlock] {
    let previousByID = Dictionary(uniqueKeysWithValues: previousBlocks.compactMap { block -> (UUID, RichBlock)? in
        guard let id = block.element?.id else { return nil }
        return (id, block)
    })

    return parsedBlocks.map { block in
        guard block.kind == .element,
              var element = block.element,
              element.isCollapsed,
              let previous = previousByID[element.id] else {
            var updated = block
            updated.children = mergeCollapsedChildren(parsedBlocks: block.children, previousBlocks: previousBlocks.flatMap(\.children))
            return updated
        }
        var updated = block
        updated.element = element
        updated.children = previous.children
        return updated
    }
}
